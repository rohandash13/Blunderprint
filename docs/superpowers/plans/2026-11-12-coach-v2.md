# Coach Mode v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give every mistake a rating-relative difficulty, let the player pick a challenge level, play every suggested move out on the board with the engine line and a plain-language eval, and clean up the voice transcript.

**Architecture:** The Python model service returns `p_best` (probability a player at this rating finds the engine move) next to `p_played`; the Java backend stores it, maps it to a level (slip/tricky/deep/unrated) and filters the report by level. The React coach adds level chips, a shared playout helper that animates the engine continuation after any move, a stepping history, and a display-only transcript cleaner.

**Tech Stack:** Python 3.11 / FastAPI / PyTorch; Java 21 / Spring Boot 3.3 / JdbcTemplate / Flyway / Testcontainers; React 18 / TypeScript / Vitest / chess.js.

**Spec:** `docs/superpowers/specs/2026-11-11-coach-v2-design.md`

## Global Constraints

- Level thresholds: slip `p_best >= 0.50`, tricky `0.15 <= p_best < 0.50`, deep `p_best < 0.15`, unrated when `p_best` is null. Config keys `report.levels.slip-min: 0.50`, `report.levels.deep-max: 0.15`.
- Default report filter: `tricky,deep`.
- Playout cap: `PLAYOUT_PLIES = 10`.
- Engine scores are centipawns from the side to move; mate is encoded as `sign * (10000 - |mate|)` (see `UciParser.toCentipawns`), so `|cp| >= 9000` means mate in `10000 - |cp|`.
- The LLM never receives FENs or UCI; move names only through `spoken` fields.
- Never print `.env` or tokens. Stage files by explicit path, never `git add -A`. Leave `scripts/hw_ceiling.py` and `results/hw_ceiling_ac.json` untracked.
- Every commit message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```python
def p_moves_batch(model, boards: list[chess.Board], moves: list[chess.Move | None],
                  ratings: list[int | None]) -> list[float | None]:
    """Probability of each given move (None stays None), one forward pass for the whole batch."""
    if not boards:
        return []
    probs = _legal_probs(model, boards, ratings)
    return [None if m is None else float(probs[i, move_index(m, b.turn == chess.BLACK)])
            for i, (b, m) in enumerate(zip(boards, moves))]


def p_played_and_best_batch(model, boards: list[chess.Board], playeds: list[chess.Move],
                            bests: list[chess.Move | None],
                            ratings: list[int | None]) -> list[tuple[float, float | None]]:
    """(p_played, p_best) per board from a single forward pass."""
    if not boards:
        return []
    probs = _legal_probs(model, boards, ratings)
    out = []
    for i, (b, p, best) in enumerate(zip(boards, playeds, bests)):
        black = b.turn == chess.BLACK
        out.append((float(probs[i, move_index(p, black)]),
                    None if best is None else float(probs[i, move_index(best, black)])))
    return out


def p_played_batch(model, boards: list[chess.Board], moves: list[chess.Move],
                   ratings: list[int | None]) -> list[float]:
    return p_moves_batch(model, boards, moves, ratings)  # type: ignore[return-value]
```

```java
package com.blunderprint.report;

import java.util.ArrayList;
import java.util.List;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

/** Maps p_best (chance a player at this rating finds the engine move) to a challenge level. */
@Component
public class Difficulty {
    public static final List<String> ALL = List.of("slip", "tricky", "deep", "unrated");
    public static final List<String> DEFAULT = List.of("tricky", "deep");

    private final double slipMin;
    private final double deepMax;

    public Difficulty(@Value("${report.levels.slip-min}") double slipMin, @Value("${report.levels.deep-max}") double deepMax) {
        this.slipMin = slipMin;
        this.deepMax = deepMax;
    }

    public String level(Double pBest) {
        if (pBest == null) return "unrated";
        if (pBest >= slipMin) return "slip";
        if (pBest < deepMax) return "deep";
        return "tricky";
    }

    /** SQL expression for the level of column {@code col}; takes two parameters: slipMin, deepMax. */
    static String sqlLevel(String col) {
        return "(case when " + col + " is null then 'unrated' when " + col + " >= ? then 'slip' when "
            + col + " < ? then 'deep' else 'tricky' end)";
    }

    double slipMin() { return slipMin; }
    double deepMax() { return deepMax; }

    /** Parses "a,b"; null means DEFAULT. Throws IllegalArgumentException on unknown or empty input. */
    public static List<String> parse(String csv) {
        if (csv == null) return DEFAULT;
        List<String> out = new ArrayList<>();
        for (String raw : csv.split(",")) {
            String s = raw.trim().toLowerCase();
            if (s.isEmpty()) continue;
            if (!ALL.contains(s)) throw new IllegalArgumentException("unknown level: " + s);
            if (!out.contains(s)) out.add(s);
        }
        if (out.isEmpty()) throw new IllegalArgumentException("no levels given");
        return out;
    }
}
```

```python
"""Break a player's mistakes down by challenge level (slip / tricky / deep / unrated).

Usage (repo root, backend running, latest run rescored so p_best is filled in):
    python scripts/difficulty_report.py --username SOME_LICHESS_USER
"""
import argparse
import json
import os
from collections import Counter, defaultdict

import requests

BASE = "http://localhost:8080"
LEVELS = ["slip", "tricky", "deep", "unrated"]


def summarize(rows: list[dict], mistake_cpl: int) -> dict:
    mistakes = [r for r in rows if r["cpl"] >= mistake_cpl]
    total = len(mistakes)
    out = {"mistakes": total, "mistake_cpl": mistake_cpl, "levels": {}}
    for level in LEVELS:
        sel = [r for r in mistakes if r["level"] == level]
        tags = Counter(t for r in sel for t in (r["tags"] or ["untagged"]))
        out["levels"][level] = {
            "count": len(sel),
            "share": round(len(sel) / total, 4) if total else 0.0,
            "avg_cpl": round(sum(r["cpl"] for r in sel) / len(sel), 1) if sel else None,
            "by_tag": dict(tags.most_common()),
        }
    return out


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--username", required=True)
    ap.add_argument("--mistake-cpl", type=int, default=100)
    ap.add_argument("--out", default=os.path.join("results", "difficulty_levels.json"))
    args = ap.parse_args()
    rows = requests.get(f"{BASE}/players/{args.username}/mistakes", timeout=60).json()
    result = {"username": args.username, **summarize(rows, args.mistake_cpl)}
    with open(args.out, "w") as f:
        json.dump(result, f, indent=2)
    for level in LEVELS:
        s = result["levels"][level]
        print(f"{level:8s} {s['count']:5d}  {s['share']:6.1%}  avg cpl {s['avg_cpl']}")


if __name__ == "__main__":
    main()
```

```ts
/** Plain-language eval from the player's side. cp uses the engine mate encoding: |cp| >= 9000 is mate in 10000-|cp|. */
export function describeEval(cpForYou: number): string {
  const abs = Math.abs(cpForYou);
  const who = cpForYou > 0 ? "You have" : "Your opponent has";
  if (abs >= 9000) return `${who} mate in ${10000 - abs}`;
  if (abs < 50) return "Roughly equal";
  const pawns = Math.round(abs / 100);
  const plural = pawns === 1 ? "pawn" : "pawns";
  if (abs < 150) return `${who} a slight edge`;
  if (abs < 300) return `${who} a clear edge, about ${pawns} ${plural}`;
  if (abs < 700) return `${who} a winning edge, about ${pawns} ${plural}`;
  return `${who} a completely winning position`;
}
```

```ts
import type { Arrow } from "./functions";

export interface Frame {
  fen: string;
  arrows: Arrow[];
}
export type StepDirection = "back" | "forward" | "start" | "end";

/** Board frames of the most recent playout, for stepping back and forward. */
export class LineHistory {
  private frames: Frame[] = [];
  private i = -1;

  reset(frame: Frame): void {
    this.frames = [frame];
    this.i = 0;
  }

  /** Continue the current line if it ends at `fen`, otherwise start a new one there. */
  begin(fen: string): void {
    if (this.last?.fen !== fen) this.reset({ fen, arrows: [] });
  }

  push(frame: Frame): void {
    this.frames = this.frames.slice(0, this.i + 1);
    this.frames.push(frame);
    this.i = this.frames.length - 1;
  }

  step(dir: StepDirection): Frame | null {
    if (this.frames.length === 0) return null;
    const lastIdx = this.frames.length - 1;
    if (dir === "back") this.i = Math.max(0, this.i - 1);
    else if (dir === "forward") this.i = Math.min(lastIdx, this.i + 1);
    else if (dir === "start") this.i = 0;
    else this.i = lastIdx;
    return this.frames[this.i];
  }

  get position(): { ply: number; total: number } {
    return { ply: Math.max(0, this.i), total: Math.max(0, this.frames.length - 1) };
  }

  get last(): Frame | null {
    return this.frames[this.frames.length - 1] ?? null;
  }
}
```

```ts
import { parseMove } from "../chess/moveParse";

export const COMMANDS = [
  "drill me", "show me", "give me a hint", "i don't know", "next one", "go back", "reset", "show me the answer",
];

export function levenshtein(a: string, b: string): number {
  const prev = Array.from({ length: b.length + 1 }, (_, j) => j);
  for (let i = 1; i <= a.length; i++) {
    let diag = prev[0];
    prev[0] = i;
    for (let j = 1; j <= b.length; j++) {
      const tmp = prev[j];
      prev[j] = Math.min(prev[j] + 1, prev[j - 1] + 1, diag + (a[i - 1] === b[j - 1] ? 0 : 1));
      diag = tmp;
    }
  }
  return prev[b.length];
}

/** Display text for a user transcript line, or null to hide it. Display only: never sent to the agent. */
export function cleanUserLine(text: string, fen: string | null): string | null {
  const trimmed = text.trim();
  if (!/[a-z0-9]/i.test(trimmed)) return null;
  if (fen) {
    const m = parseMove(fen, trimmed);
    if (m) return m.san;
  }
  const norm = trimmed.toLowerCase().replace(/[^a-z' ]/g, "").replace(/\s+/g, " ").trim();
  if (norm.split(" ").length <= 4) {
    let best: { cmd: string; d: number } | null = null;
    for (const cmd of COMMANDS) {
      if (cmd[0] !== norm[0]) continue; // a different first sound is a different phrase
      const d = levenshtein(norm, cmd);
      if (!best || d < best.d) best = { cmd, d };
    }
    if (best && best.d > 0 && best.d <= 3 && best.d <= 0.4 * best.cmd.length) return best.cmd;
    if (best && best.d === 0) return best.cmd;
  }
  return trimmed;
}
```
